using System; using System.Collections; using System.Collections.Generic; using System.Drawing; using System.Globalization; using System.IO; using System.Linq; using System.Text; using Rhino; using Rhino.Geometry; using Grasshopper; using Grasshopper.Kernel; using Grasshopper.Kernel.Data; using Grasshopper.Kernel.Types; public class Script_Instance : GH_ScriptInstance { private void RunScript(List M, string P, string T, string N, bool W, ref object A, ref object B) { if (M == null || M.Count == 0) { B = "No meshes supplied."; return; } if (!W) { B = "Set W = true to export."; return; } if (string.IsNullOrWhiteSpace(P)) { B = "No output path (P) supplied."; return; } string title = string.IsNullOrWhiteSpace(T) ? "3D Viewer" : T; const int maxVertsPerChunk = 60000; // safety margin under the 65536 (ushort) index limit var chunks = new List(); var partNames = new List(); var bbox = BoundingBox.Empty; for (int mi = 0; mi < M.Count; mi++) { var mesh = M[mi]; if (mesh == null || mesh.Vertices.Count == 0) continue; mesh.Faces.ConvertQuadsToTriangles(); if (mesh.Normals.Count != mesh.Vertices.Count) mesh.Normals.ComputeNormals(); bbox.Union(mesh.GetBoundingBox(false)); int partIndex = partNames.Count; partNames.Add("Part " + (partIndex + 1)); foreach (var chunk in SplitMesh(mesh, maxVertsPerChunk)) { chunk.Part = partIndex; chunks.Add(chunk); } } if (chunks.Count == 0) { B = "Nothing to export after processing."; return; } string sceneJson = BuildSceneJson(chunks, bbox, partNames); string sceneB64 = Convert.ToBase64String(Encoding.UTF8.GetBytes(sceneJson)); string noteB64 = Convert.ToBase64String(Encoding.UTF8.GetBytes(N ?? "")); string html = BuildHtml(title, sceneB64, noteB64); File.WriteAllText(P, html, new UTF8Encoding(false)); long bytes = new FileInfo(P).Length; int totalVerts = chunks.Sum(c => c.VertexCount); int totalTris = chunks.Sum(c => c.TriCount); A = P; B = $"{M.Count} mesh(es), {totalVerts:N0} verts, {totalTris:N0} tris, {FormatBytes(bytes)}"; } // --------------------------------------------------------------- // Mesh -> chunk splitting // --------------------------------------------------------------- private class Chunk { public List Positions = new List(); // xyz interleaved public List Normals = new List(); // xyz interleaved public List Indices = new List(); public int Part; public int VertexCount => Positions.Count / 3; public int TriCount => Indices.Count / 3; } private List SplitMesh(Mesh mesh, int maxVerts) { var faces = mesh.Faces; var verts = mesh.Vertices; var norms = mesh.Normals; var chunk = new Chunk(); var remap = new Dictionary(); // original vertex index -> local index var result = new List(); foreach (var f in faces) { int[] tri = f.IsTriangle ? new[] { f.A, f.B, f.C } : new[] { f.A, f.B, f.C, f.C, f.D, f.A }; // defensive; quads are converted above for (int t = 0; t < tri.Length; t += 3) { int newCount = 0; for (int k = 0; k < 3; k++) if (!remap.ContainsKey(tri[t + k])) newCount++; if (chunk.VertexCount + newCount > maxVerts && chunk.VertexCount > 0) { result.Add(chunk); chunk = new Chunk(); remap.Clear(); } for (int k = 0; k < 3; k++) { int orig = tri[t + k]; if (!remap.TryGetValue(orig, out int local)) { local = chunk.VertexCount; remap[orig] = local; var p = verts[orig]; var n = norms.Count == verts.Count ? norms[orig] : new Vector3f(0, 0, 1); chunk.Positions.Add(p.X); chunk.Positions.Add(p.Y); chunk.Positions.Add(p.Z); chunk.Normals.Add(n.X); chunk.Normals.Add(n.Y); chunk.Normals.Add(n.Z); } chunk.Indices.Add(local); } } } if (chunk.VertexCount > 0) result.Add(chunk); return result; } private string FormatBytes(long bytes) { string[] units = { "B", "KB", "MB", "GB" }; double v = bytes; int u = 0; while (v >= 1024 && u < units.Length - 1) { v /= 1024; u++; } return $"{v:0.##} {units[u]}"; } // --------------------------------------------------------------- // Scene JSON (hand-written, no JSON library dependency) // --------------------------------------------------------------- private string F(float v) => v.ToString("0.###", CultureInfo.InvariantCulture); private string JsonEsc(string s) => (s ?? "").Replace("\\", "\\\\").Replace("\"", "\\\""); private string BuildSceneJson(List chunks, BoundingBox bbox, List partNames) { var sb = new StringBuilder(); sb.Append('{'); sb.Append("\"bounds\":{") .Append("\"min\":[").Append(F((float)bbox.Min.X)).Append(',').Append(F((float)bbox.Min.Y)).Append(',').Append(F((float)bbox.Min.Z)).Append("],") .Append("\"max\":[").Append(F((float)bbox.Max.X)).Append(',').Append(F((float)bbox.Max.Y)).Append(',').Append(F((float)bbox.Max.Z)).Append("]") .Append("},"); sb.Append("\"parts\":["); for (int i = 0; i < partNames.Count; i++) { if (i > 0) sb.Append(','); sb.Append('"').Append(JsonEsc(partNames[i])).Append('"'); } sb.Append("],"); sb.Append("\"chunks\":["); for (int i = 0; i < chunks.Count; i++) { var c = chunks[i]; if (i > 0) sb.Append(','); sb.Append('{'); sb.Append("\"part\":").Append(c.Part).Append(','); sb.Append("\"pos\":[").Append(string.Join(",", c.Positions.Select(F))).Append("],"); sb.Append("\"nrm\":[").Append(string.Join(",", c.Normals.Select(F))).Append("],"); sb.Append("\"idx\":[").Append(string.Join(",", c.Indices)).Append(']'); sb.Append('}'); } sb.Append(']'); sb.Append('}'); return sb.ToString(); } // --------------------------------------------------------------- // HTML + WebGL viewer template // --------------------------------------------------------------- private string BuildHtml(string title, string sceneB64, string noteB64) { string safeTitle = System.Net.WebUtility.HtmlEncode(title); string html = @" @@TITLE@@
